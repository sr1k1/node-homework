# Task 1: Find the total price of each of the first five orders, ordered by order_id
SELECT orders.order_id AS order_id, SUM(products.price * line_items.quantity) AS total_price FROM orders 
INNER JOIN line_items 
ON orders.order_id = line_items.order_id 
INNER JOIN products 
ON line_items.product_id = products.product_id 
GROUP BY orders.order_id 
ORDER BY orders.order_id 
LIMIT 5; 

# Task 2: Understanding Subqueries
SELECT customers.customer_name AS customer_name, AVG(t.total_price) AS average_order_price FROM customers 
INNER JOIN 
(SELECT orders.customer_id AS customer_id, orders.order_id AS order_id, SUM(products.price * line_items.quantity) AS total_price FROM orders 
INNER JOIN line_items 
ON orders.order_id = line_items.order_id 
INNER JOIN products 
ON line_items.product_id = products.product_id 
GROUP BY orders.order_id) 
AS t 
ON customers.customer_id = t.customer_id 
GROUP BY customers.customer_id, customers.customer_name 
ORDER BY customers.customer_name;

# Task 4: Aggregation with Having
SELECT employees.first_name, employees.last_name, COUNT(orders.order_id) AS order_count FROM employees 
INNER JOIN orders 
ON employees.employee_id = orders.employee_id 
GROUP BY employees.employee_id, employees.first_name, employees.last_name 
HAVING COUNT(order_id) > 5 
ORDER BY employees.last_name;
